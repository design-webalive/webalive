class Component extends DCLogic {
  state = { view: 'queue', filter: 'All', selected: ['0039'], rows: ROWS, t1: false, t2: false, sig: '', signed: null, signedAt: '', sigName: '', err: false, toast: null, toastTone: 'success', menu: null };

  printDoc = (mode, ref) => {
    const name = (ref || 'STT-2026-0039') + '-Raghunathan';
    const node = document.querySelector('[data-printarea]');
    if (mode === 'word') {
      const html = '<html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:w="urn:schemas-microsoft-com:office:word">'
        + '<head><meta charset="utf-8"><title>' + name + '</title></head><bo' + 'dy>'
        + (node ? node.innerHTML : '') + '</bo' + 'dy></html>';
      const url = URL.createObjectURL(new Blob(['\ufeff' + html], { type: 'application/msword' }));
      const a = document.createElement('a');
      a.href = url; a.download = name + '.doc';
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 5000);
      this.flash(name + '.doc downloaded.');
      return;
    }
    this.flash('Opening the print view — choose "Save as PDF" to keep a copy.', 'info');
    setTimeout(() => window.print(), 250);
  };

  flash(msg, tone) {
    clearTimeout(this._t);
    this.setState({ toast: msg, toastTone: tone || 'success' });
    this._t = setTimeout(() => this.setState({ toast: null }), 3600);
  }

  openMenu = (e, id) => {
    const r = e.currentTarget.getBoundingClientRect();
    const x = Math.min(r.right - 168, window.innerWidth - 178);
    const y = Math.min(r.bottom + 6, window.innerHeight - 130);
    this.setState({ menu: { id: id, x: Math.max(8, x), y: Math.max(8, y) } });
  };

  setStatus(id, status, note) {
    this.setState(s => ({ rows: s.rows.map(r => r.id === id ? { ...r, status, note } : r) }));
  }

  confirm = () => {
    const name = this.state.sig.trim();
    if (!this.state.t1 || !this.state.t2 || !name) { this.setState({ err: true }); return; }
    const at = 'Wed, 8 Jul 2026 09:12';
    this.setState({ err: false, signed: 'approved', signedAt: at, sigName: name });
    this.setStatus('0039', 'Approved', 'Confirmed by ' + name + ', 8 Jul 09:12');
  };

  decline = () => {
    const name = this.state.sig.trim() || 'Daniel Orsini';
    this.setState({ err: false, signed: 'declined', sigName: name });
    this.setStatus('0039', 'Declined', 'Declined by ' + name + ', 8 Jul 09:12');
  };

  renderVals() {
    const s = this.state;
    const rec = s.rows.find(r => r.id === '0039');
    const awaiting = s.rows.filter(r => r.status === 'Awaiting sign-off').length;
    const rows = s.filter === 'All' ? s.rows : s.rows.filter(r => r.status === s.filter);
    const img = (w) => React.createElement('img', { src: (window.__resources && window.__resources.logoMark) || 'assets/logo-mark.png', width: w, alt: 'WebCommander' });
    const DS = window.WebCommanderDesignSystem_d41e75 || {};
    const el = (n, p, c) => DS[n] ? React.createElement(DS[n], p, c) : null;

    const timeline = [
      { label: 'Application submitted and rendered to PDF', when: 'Tue, 23 Jun 2026 14:08' },
      { label: 'Sign-off request emailed to d.orsini@holmesglen.edu.au', when: 'Tue, 23 Jun 2026 14:08' },
      { label: 'Reminder 1 of 3 sent', when: 'Fri, 26 Jun 2026 09:00' },
      { label: 'Reminder 2 of 3 sent', when: 'Mon, 6 Jul 2026 09:00' }
    ];
    if (rec.status === 'Awaiting sign-off') timeline.push({ label: 'Expires with no sign-off', when: '5pm, Wed 8 Jul 2026 (closing date)' });
    if (rec.status === 'Approved') timeline.push({ label: 'Confirmed by ' + (s.sigName || 'Daniel Orsini') + ' — applicant and award team notified', when: s.signedAt || 'Wed, 8 Jul 2026 09:12' });
    if (rec.status === 'Declined') timeline.push({ label: 'Declined by ' + (s.sigName || 'Daniel Orsini') + ' — applicant notified', when: 'Wed, 8 Jul 2026 09:12' });

    const cell = (t, c, w) => React.createElement('div', { style: { font: (w || 400) + ' 13px/1.4 var(--wc-font-sans)', color: c || 'var(--wc-text-secondary)' } }, t);
    const queueTable = {
      selectable: true, rowActions: false, gutter: 30, rowHeight: 66,
      rows: rows, selected: s.selected, onSelect: (n) => this.setState({ selected: n }),
      columns: [
        { key: 'ref', header: 'Reference', width: 'minmax(150px,1.1fr)', render: (r) => React.createElement('a', {
            href: '#' + r.ref, onClick: (e) => { e.preventDefault(); if (r.id === '0039') this.setState({ view: 'record' }); else this.flash('Only STT-2026-0039 is populated in this prototype.', 'info'); },
            style: { font: 'var(--wc-weight-semibold) 13px/1.4 var(--wc-font-sans)' } }, r.ref) },
        { key: 'applicant', header: 'Applicant', width: 'minmax(180px,1.5fr)', render: (r) => React.createElement('div', null,
            cell(r.applicant, 'var(--wc-text)', 'var(--wc-weight-semibold)'), cell(r.cat, 'var(--wc-text-micro)')) },
        { key: 'org', header: 'Employer or RTO', width: 'minmax(190px,1.6fr)', render: (r) => React.createElement('div', null,
            cell(r.org, 'var(--wc-text)'), cell(r.contact, 'var(--wc-text-micro)')) },
        { key: 'submitted', header: 'Submitted', width: 'minmax(150px,1.2fr)', render: (r) => cell(r.submitted, 'var(--wc-text-muted)') },
        { key: 'status', header: 'Status', width: 'minmax(140px,1.1fr)', render: (r) => el('StatusPill', { variant: 'filled', tone: TONE[r.status] }, r.status) },
        { key: 'note', header: 'Sign-off activity', width: 'minmax(190px,1.6fr)', render: (r) => cell(r.note, 'var(--wc-text-micro)') },
        { key: 'act', header: '', width: '130px', render: (r) => React.createElement('div',
            { style: { display: 'flex', alignItems: 'center', gap: 4, justifyContent: 'flex-end' } },
            el('Button', { size: 'sm', variant: 'link', icon: 'printer', onClick: () => this.printDoc('pdf', r.ref) }, 'Print'),
            React.createElement('button', {
              type: 'button', 'aria-label': 'More', onClick: (e) => this.openMenu(e, r.id),
              style: { width: 28, height: 28, border: 0, borderRadius: '50%', background: 'transparent', cursor: 'pointer', color: 'var(--wc-text-micro)', fontSize: 15, lineHeight: 1 }
            }, '\u22EE')) }
      ]
    };

    return {
      true: true,
      viewOptions: VIEWS.map(v => v[1]),
      viewLabel: (VIEWS.find(v => v[0] === s.view) || VIEWS[0])[1],
      setViewByLabel: (label) => this.setState({ view: (VIEWS.find(v => v[1] === label) || VIEWS[0])[0] }),
      resetDemo: () => this.setState({ rows: ROWS, view: 'queue', filter: 'All', selected: ['0039'], t1: false, t2: false, sig: '', signed: null, sigName: '', err: false }),
      isAdmin: s.view === 'queue' || s.view === 'record',
      isQueue: s.view === 'queue', isRecord: s.view === 'record', isEmail: s.view === 'email', isSignoff: s.view === 'signoff',
      adminTitle: React.createElement('span', { style: { whiteSpace: 'nowrap' } }, s.view === 'record' ? 'Award application' : 'Submissions'),
      logo: img(30), railLogo: img(32), credits: { used: 2, total: 5 },
      topRight: React.createElement(React.Fragment, null,
        el('Button', { variant: 'neutral', size: 'sm', icon: 'external-link' }, 'View Site'),
        el('IconButton', { icon: 'bell', variant: 'plain', badge: awaiting, label: 'Notifications' })),
      chaseTitle: awaiting === 1 ? '1 entry is waiting on employer sign-off' : awaiting + ' entries are waiting on employer sign-off',
      chaseAction: el('Button', { size: 'sm', variant: 'secondary', onClick: () => this.setState({ filter: 'Awaiting sign-off' }) }, 'Show awaiting'),
      queueTabs: el('Tabs', { value: 'Submissions', items: ['Submissions'], style: { border: 'none' } }),
      countLabel: s.rows.length + ' entries · ' + s.rows.filter(r => r.status === 'Approved').length + ' approved',
      filterOptions: ['All', 'Awaiting sign-off', 'Approved', 'Declined', 'Expired'],
      filter: s.filter, setFilter: (f) => this.setState({ filter: f }),
      selectionLabel: s.selected.length ? s.selected.length + ' selected' : 'Nothing selected',
      queueTable: queueTable,
      printSelected: () => this.state.selected.length
        ? this.printDoc('pdf')
        : this.flash('Select at least one entry to print.', 'info'),
      recStatus: rec.status, recTone: TONE[rec.status],
      timeline: timeline,
      files: [
        { kind: 'PDF', name: 'STT-2026-0039-Raghunathan.pdf', size: '4 pages · 1.1 MB' },
        { kind: 'JPG', name: 'main-wagyu-striploin.jpg', size: '2.4 MB' },
        { kind: 'JPG', name: 'dessert-rice-quandong.jpg', size: '1.9 MB' }
      ],
      dlPdf: () => this.printDoc('pdf'),
      dlWord: () => this.printDoc('word'),
      menu: !!s.menu,
      menuStyle: s.menu
        ? 'position:fixed;z-index:61;top:' + s.menu.y + 'px;left:' + s.menu.x + 'px;min-width:168px;padding:5px 0;background:#fff;border:1px solid var(--wc-border);border-radius:6px;box-shadow:0 8px 24px rgba(16,24,40,.12)'
        : 'display:none',
      closeMenu: () => this.setState({ menu: null }),
      openRecordMenu: (e) => this.openMenu(e, '0039'),
      menuDetails: () => { const id = s.menu && s.menu.id; this.setState({ menu: null }); if (id === '0039') this.setState({ view: 'record' }); else this.flash('Only STT-2026-0039 is populated in this prototype.', 'info'); },
      menuPdf: () => { const id = s.menu && s.menu.id; this.setState({ menu: null }); this.printDoc('pdf', 'STT-2026-' + id); },
      menuWord: () => { const id = s.menu && s.menu.id; this.setState({ menu: null }); this.printDoc('word', 'STT-2026-' + id); },
      resend: () => this.flash('Sign-off request resent to d.orsini@holmesglen.edu.au.'),
      noop: (e) => { e.preventDefault(); this.printDoc('pdf'); },
      goQueue: () => this.setState({ view: 'queue' }),
      goEmail: () => this.setState({ view: 'email' }),
      goSignoff: () => this.setState({ view: 'signoff' }),
      t1: s.t1, t2: s.t2, err: s.err,
      toggleT1: () => this.setState(p => ({ t1: !p.t1, err: false })),
      toggleT2: () => this.setState(p => ({ t2: !p.t2, err: false })),
      onSig: (e) => this.setState({ sig: e.target.value, err: false }),
      confirmSignoff: this.confirm, declineSignoff: this.decline,
      notSigned: !s.signed, isApprovedView: s.signed === 'approved', isDeclinedView: s.signed === 'declined',
      sigName: s.sigName, signedAt: s.signedAt,
      toast: s.toast, toastTone: s.toastTone
    };
  }
}
</script>
